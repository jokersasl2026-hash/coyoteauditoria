    <script>
        // ============================================================================
        // CONFIGURACIÓN DE TU PROYECTO DE FIREBASE
        // ============================================================================
        // Debes reemplazar esta URL por la de tu base de datos de la consola Firebase
        const firebaseConfig = {
            databaseURL: "https://console.firebase.google.com/project/coyoteauditoria/database/coyoteauditoria-default-rtdb/data/~2F"
        };
        
        // Inicializa la conexión web con Firebase
        firebase.initializeApp(firebaseConfig);
        const dbRef = firebase.database().reference('/sala_monitoreo');

        // Función para dar formato de dinero en Guaraníes (Gs. 1.500.000)
        function formatearMoneda(valor) {
            if (valor === undefined || valor === null) return "Gs. 0";
            return "Gs. " + Number(valor).toLocaleString('de-DE');
        }
        // Escucha los cambios en la base de datos en tiempo real
        dbRef.on('value', (snapshot) => {
            const data = snapshot.val();
            if (!data) return;

            // 1. Actualizar los contenedores globales principales
            document.getElementById('txt-reloj').innerText = data.ultima_actualizacion || '---';
            document.getElementById('neto-global').innerText = formatearMoneda(data.neto_global);
            document.getElementById('total-in').innerText = formatearMoneda(data.total_in_global);
            document.getElementById('total-out').innerText = formatearMoneda(data.total_out_global);

            // 2. Limpiar el contenedor para reconstruir la lista de las máquinas de la sala
            const contenedor = document.getElementById('contenedor-maquinas');
            contenedor.innerHTML = '';

            if (data.maquinas) {
                // Ordenar las máquinas numéricamente por su nombre oficial
                const maquinasOrdenadas = Object.keys(data.maquinas).sort((a, b) => {
                    const numA = parseInt(a.replace('maq_', '')) || 0;
                    const numB = parseInt(b.replace('maq_', '')) || 0;
                    return numA - numB;
                });

                maquinasOrdenadas.forEach((key) => {
                    const m = data.maquinas[key];
                    const nombreOficial = key.replace('maq_', '');
                    
                    // Determinar si la placa está ONLINE según el indicador de Tkinter de la PC
                    const esOnline = m.status === '[ON]';
                    const claseStatus = esOnline ? 'online' : 'offline';

                    // Crear el bloque visual para cada máquina
                    const itemHtml = `
                        <div class="item-maquina">
                            <div class="info-izq">
                                <div class="status-dot ${claseStatus}"></div>
                                <span class="nombre-maq">Máquina ${nombreOficial}</span>
                            </div>
                            <div class="valores-der">
                                <div class="val-in">IN: ${formatearMoneda(m.in)}</div>
                                <div class="val-out">OUT: ${formatearMoneda(m.out)}</div>
                            </div>
                        </div>
                    `;
                    contenedor.innerHTML += itemHtml;
                });
            } else {
                contenedor.innerHTML = '<p style="text-align: center; color: #7f8c8d; padding: 20px;">No se encontraron máquinas activas.</p>';
            }
        }, (error) => {
            console.error("Error leyendo datos de Firebase:", error);
        });
    </script>
